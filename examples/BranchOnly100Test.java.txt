package mn.edu.must.sqat;

import static org.junit.jupiter.api.Assertions.assertEquals;
import static org.junit.jupiter.api.Assertions.assertThrows;

import org.junit.jupiter.api.Test;

/**
 * Counter-example for B5: these 4 bonus tests (+ GcdTest) give JaCoCo BRANCH 16/16,
 * yet 5 feasible DU pairs of bonus() stay uncovered:
 * score(49->54), score(51->56), present(35->50T), present(40->50F), best(43->53F).
 */
class BranchOnly100Test {

    @Test
    void x1_null_throws() { // 32a T
        assertThrows(IllegalArgumentException.class, () -> DataFlowTargets.bonus(null, 0));
    }

    @Test
    void x2_negativeThreshold_throws() { // 32a F, 32b T
        assertThrows(IllegalArgumentException.class, () -> DataFlowTargets.bonus(new int[] {}, -1));
    }

    @Test
    void x3_streak5_thenBreak_returns8() { // 32b F, 38 T/F, 39 T/F, 42 T/F, 50 T, 53 T
        assertEquals(8, DataFlowTargets.bonus(new int[] {1, 1, 1, 1, 1, 0, 1}, 3));
    }

    @Test
    void x4_singleAbsent_returns0() { // 50 F, 53 F
        assertEquals(0, DataFlowTargets.bonus(new int[] {0}, 1));
    }
}
